print(f"Intercept: {intercept:.10f}")

def format_duration(seconds):
    mins = seconds / 60
    hours = mins / 60
    days = hours / 24
    return f"{seconds:.2f}s ({mins:.2f}m, {hours:.2f}h, {days:.2f}d)"

for size in [5_000_000, 5_000_000_000]:
